# Nightshade OS GRUB theme.
#
# Flat #161226 field, the Nightshade lockup centred above the menu, cool
# off-white menu text with a muted violet selection. No generated artwork: the
# lockup PNG is the only image asset and it is used as-is.
#
# Fonts are produced at build time by hook 0800 (grub-mkfont over DejaVu Sans
# Mono, renamed to "Nightshade Mono") and loaded by grub.cfg before the theme
# is applied.
#
# Two things about GRUB's gfxmenu that this file depends on:
#
#   1. In an `image` component, `file` is what triggers the bitmap load and
#      scale. Geometry declared AFTER `file` resizes the component box without
#      rescaling the bitmap, so the logo renders at the wrong size. left/top/
#      width/height must come first.
#
#   2. The lockup PNG has an opaque dark plate, not a transparent background,
#      so its box paints over anything beneath it. The logo and the menu must
#      not overlap -- an overlapping logo silently truncates the menu labels
#      rather than blending with them.

title-text: ""
desktop-color: "#161226"
message-color: "#E6E4F0"
message-bg-color: "#161226"
terminal-font: "Nightshade Mono Regular 16"
terminal-left: "8%"
terminal-top: "8%"
terminal-width: "84%"
terminal-height: "84%"

+ image {
    id       = "logo"
    # No width/height here on purpose: this gfxmenu ignores both on an image
    # component and instead lays the image out at its natural size, clamped to
    # the space remaining to the right of `left`. So the asset itself is what
    # controls the size. logo.png is generated by hook 0800 at exactly 440x120
    # from the 1204x328 source lockup, which preserves its 3.67 aspect and lets
    # GRUB blit it 1:1 -- sharper than anything GRUB's own scaler produces.
    #
    # 50%-220 centres a 440px-wide image at any resolution >= 880 wide.
    left     = 50%-220
    top      = 12%
    file     = "logo.png"
}

+ boot_menu {
    # Starts at 38% of 768 = 292px, comfortably below the logo box, which ends
    # at 12% + 120 = 212px. See note 2 above.
    left                 = 50%-260
    top                  = 38%
    width                = 51%
    height               = 32%

    item_font            = "Nightshade Mono Regular 16"
    selected_item_font   = "Nightshade Mono Regular 16"
    # Unselected sits well back from the background; selected is the violet
    # picked out of the logo's cube so the highlight reads as brand, not alert.
    item_color           = "#9A93B8"
    selected_item_color  = "#C4B5FD"

    item_height          = 30
    item_spacing         = 8
    # No per-entry icons; --class nightshade has no icon set, and the default
    # icon gutter would otherwise indent every label by ~50px.
    icon_width           = 0
    icon_height          = 0
    item_icon_space      = 0
}

+ progress_bar {
    # The magic id makes GRUB drive this from the countdown automatically, and
    # hide it as soon as a keypress cancels the countdown.
    id           = "__timeout__"
    left         = 50%-260
    top          = 78%
    width        = 520
    height       = 18
    show_text    = true
    text         = "@TIMEOUT_NOTIFICATION_SHORT@"
    font         = "Nightshade Mono Regular 16"
    # GRUB centres the countdown text over the bar, so it crosses both the fill
    # and the empty track as the bar drains. A deep violet fill (rather than the
    # bright selection violet) keeps one light text colour legible against both.
    text_color   = "#E6E4F0"
    fg_color     = "#6E5AAE"
    bg_color     = "#241C3C"
    border_color = "#332A52"
}

+ label {
    id     = "hint"
    text   = "Enter  boot        E  edit        C  console"
    left   = 0
    top    = 88%
    width  = 100%
    align  = "center"
    font   = "Nightshade Mono Regular 16"
    color  = "#6F6890"
}
